# --- PDF reading ---
pymupdf                 # reads text out of PDF files (imported as "fitz")

# --- RAG framework ---
langchain               # the orchestration framework (glue between the pieces)
langchain-community     # community integrations (loaders, etc.)
langchain-huggingface   # lets LangChain use local SentenceTransformers embeddings
langchain-chroma        # lets LangChain talk to the Chroma vector database
langchain-groq          # lets LangChain call the Groq LLM

# --- embeddings + vector store engines ---
sentence-transformers   # runs the local BAAI/bge-small-en-v1.5 embedding model
chromadb                # the actual vector database engine

# --- UI + config ---
streamlit               # the web UI
python-dotenv           # loads your secret API key from a .env file

# --- Database ---
pymongo                 # MongoDB driver for Python
